/*******************************************************************************
 *                                                                      *
 * Formula Name : ORA_WFM_TCR_DAILY_OT_PREMIUM_SPLIT_AP                *
 *                                                                      *
 * Formula Type : Time Calculation Rules                                *
 *                                                                      *
 * Description  : Splits daily worked hours into regular, OT, and       *
 *                premium OT buckets.                                   *
 *                                                                      *
 * Detail       : Returns hours up to 8 in the under-threshold bucket,  *
 *                the next 2 hours in the OT bucket, and any hours      *
 *                above 10 in the premium OT bucket. The formula        *
 *                assumes detail entries are processed in time order     *
 *                and uses end-of-day / end-of-period markers to reset   *
 *                the daily accumulator.                                 *
 *                                                                      *
 * Rule Parameters : None. Thresholds are fixed at 8 and 10.            *
 *                                                                      *
 * Change History                                                       *
 * --------------                                                       *
 * Who                          Ver     Date           Description      *
 * ---------------------------   ------  ------------   --------------- *
 * Time and Labor Formula AI Agent 1.0   2026-MAY-07    Created.        *
 *                                                                      *
 ******************************************************************************/

/* Block 1: Default Declarations */
DEFAULT FOR HWM_CTXARY_RECORD_POSITIONS IS EMPTY_TEXT_NUMBER
DEFAULT FOR measure IS EMPTY_NUMBER_NUMBER

/* Block 2: Inputs */
INPUTS ARE
    HWM_CTXARY_RECORD_POSITIONS,
    measure

/* Block 3: Context and Parameter Setup */
ffs_id = GET_CONTEXT(HWM_FFS_ID, 0)
rule_id = GET_CONTEXT(HWM_RULE_ID, 0)
ffName = 'ORA_WFM_TCR_DAILY_OT_PREMIUM_SPLIT_AP'
rLog = add_rlog(ffs_id, rule_id, '>>> Enter - ' || ffName)

NullText = '**FF_NULL**'
RecPositionEndPeriod = 'END_PERIOD'
RecPositionEndDay = 'END_DAY'
RecPositionDetail = 'DETAIL'

hSumLvl = Get_Hdr_Text(rule_id, 'RUN_SUMMATION_LEVEL', 'TIMECARD')
hExecType = Get_Hdr_Text(rule_id, 'RULE_EXEC_TYPE', 'CREATE')

rLog = add_rlog(
    ffs_id,
    rule_id,
    'Rule Header: ffs_id=' || TO_CHAR(ffs_id) ||
    ' , rule_id=' || TO_CHAR(rule_id) ||
    ' , hSumLvl=' || hSumLvl ||
    ' , hExecType=' || hExecType
)

thresholdUnder = 8
thresholdPremium = 10

rLog = add_rlog(
    ffs_id,
    rule_id,
    'Thresholds: under=' || TO_CHAR(thresholdUnder) ||
    ' , premium=' || TO_CHAR(thresholdPremium)
)

/* Block 4: Processing Logic */
OUT_MEASURE_UNDER = EMPTY_NUMBER_NUMBER
OUT_MEASURE_OT = EMPTY_NUMBER_NUMBER
OUT_MEASURE_PREMIUM = EMPTY_NUMBER_NUMBER

dayWorkedHrs = 0
max_loop = 1000
tcRecordCount = HWM_CTXARY_RECORD_POSITIONS.count

nidx = 0

WHILE (nidx < tcRecordCount) LOOP
(
    nidx = nidx + 1

    IF (nidx > max_loop) THEN
    (
        ex = RAISE_ERROR(
                 ffs_id,
                 rule_id,
                 'Formula ' || ffName || ' terminated due to possible end-less loop.'
             )
    )

    tcMeasure = 0
    tcRecPos = NullText

    IF (measure.exists(nidx)) THEN
    (
        tcMeasure = measure[nidx]
    )

    IF (HWM_CTXARY_RECORD_POSITIONS.exists(nidx)) THEN
    (
        tcRecPos = HWM_CTXARY_RECORD_POSITIONS[nidx]
    )

    rLog = add_rlog(
        ffs_id,
        rule_id,
        'Record ' || TO_CHAR(nidx) ||
        ' , recPos=' || tcRecPos ||
        ' , measure=' || TO_CHAR(tcMeasure) ||
        ' , dayWorkedHrs=' || TO_CHAR(dayWorkedHrs)
    )

    IF (UPPER(tcRecPos) = RecPositionDetail) THEN
    (
        allocUnder = 0
        allocOt = 0
        allocPremium = 0

        IF (tcMeasure > 0) THEN
        (
            remainingUnder = thresholdUnder - dayWorkedHrs
            IF (remainingUnder < 0) THEN
            (
                remainingUnder = 0
            )

            allocUnder = LEAST(tcMeasure, remainingUnder)
            remAfterUnder = tcMeasure - allocUnder

            remainingOtCap = thresholdPremium - dayWorkedHrs - allocUnder
            IF (remainingOtCap < 0) THEN
            (
                remainingOtCap = 0
            )

            allocOt = LEAST(remAfterUnder, remainingOtCap)
            allocPremium = remAfterUnder - allocOt

            IF (allocPremium < 0) THEN
            (
                allocPremium = 0
            )

            dayWorkedHrs = dayWorkedHrs + tcMeasure
        )

        OUT_MEASURE_UNDER[nidx] = allocUnder
        OUT_MEASURE_OT[nidx] = allocOt
        OUT_MEASURE_PREMIUM[nidx] = allocPremium

        rLog = add_rlog(
            ffs_id,
            rule_id,
            'Bucket split: under=' || TO_CHAR(allocUnder) ||
            ' , ot=' || TO_CHAR(allocOt) ||
            ' , premium=' || TO_CHAR(allocPremium) ||
            ' , runningDay=' || TO_CHAR(dayWorkedHrs)
        )
    )

    IF (
           UPPER(tcRecPos) = RecPositionEndDay
        OR UPPER(tcRecPos) = RecPositionEndPeriod
       ) THEN
    (
        rLog = add_rlog(
            ffs_id,
            rule_id,
            'End of day/period marker encountered. Day accumulator reset.'
        )
        dayWorkedHrs = 0
    )
)

rLog = add_rlog(ffs_id, rule_id, '<<< Exit - ' || ffName)

/* Block 5: Outputs */
RETURN OUT_MEASURE_UNDER,
       OUT_MEASURE_OT,
       OUT_MEASURE_PREMIUM
