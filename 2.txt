/* +======================================================================+
 *
 * Formula Name : ORA_WFM_TCR_DAILY_OT_SPLIT_AP
 *
 * Formula Type:  Time Calculation Rules
 *
 * Description:   Splits a daily measure into an under-threshold bucket,
 *                an overtime bucket for the first 2 hours above the
 *                daily threshold, and a premium overtime bucket for any
 *                hours above the premium threshold.
 *
 * Detail:        Hours up to the daily OT threshold are returned as-is.
 *                The next 2 hours are returned in the OT bucket.
 *                Any remaining hours above the premium threshold are
 *                returned in the premium OT bucket.
 *                If an END_DAY or END_PERIOD row has a zero measure, the
 *                formula falls back to HWM_CTXARY_HWM_MEASURE_DAY.
 *
 * Rule Parameters (example names expected on rule):
 *   DEFINED_LIMIT (Number) - Daily OT threshold; default 8
 *   Premium threshold is calculated as DEFINED_LIMIT + 2
 *
 * Change History
 * --------------
 * Who                          Ver      Date           Description
 *-----------------             ------   ------------   -----------------------------------------
 * Time and Labor Formula AI Agent
 *                              1.0      2026-MAY-07   Initial development per user request
 *
 ***************************************************************************/

/*
 * Block 1: Default Declarations
 */
DEFAULT FOR HWM_CTXARY_RECORD_POSITIONS IS EMPTY_TEXT_NUMBER
DEFAULT FOR HWM_CTXARY_HWM_MEASURE_DAY IS EMPTY_NUMBER_NUMBER
DEFAULT FOR measure IS EMPTY_NUMBER_NUMBER
DEFAULT FOR StartTime IS EMPTY_DATE_NUMBER

/*
 * Block 2: Inputs
 */
INPUTS ARE
  HWM_CTXARY_RECORD_POSITIONS,
  HWM_CTXARY_HWM_MEASURE_DAY,
  measure,
  StartTime

/*
 * Block 3: Context and Parameter Setup
 */
ffName = 'ORA_WFM_TCR_DAILY_OT_SPLIT_AP -v1.0 '
ffs_id = GET_CONTEXT(HWM_FFS_ID, 0)
rule_id = GET_CONTEXT(HWM_RULE_ID, 0)
rLog = add_rlog (ffs_id, rule_id, '>>> Enter - ' || ffName )

NullDate = '01-JAN-1900'(DATE)
NullDateTime = '1900/01/01 00:00:00'(DATE)
NullText = '**FF_NULL**'
sTrue = 'Y'
sFalse = 'N'

RecPositionEndPeriod = 'END_PERIOD'
RecPositionEndDay = 'END_DAY'
RecPositionDetail = 'DETAIL'

hSumLvl = Get_Hdr_Text(rule_id, 'RUN_SUMMATION_LEVEL', 'TIMECARD')
hExecType = Get_Hdr_Text(rule_id, 'RULE_EXEC_TYPE', 'CREATE')

hCreateYn = sFalse
if (upper(hExecType) = 'CREATE') then (
  hCreateYn = sTrue
)

dailyThreshold = get_rvalue_number (rule_id, 'DEFINED_LIMIT', 8)
premiumThreshold = dailyThreshold + 2

rLog = add_rlog (ffs_id, rule_id, 'Rule Header and Parameters: ' ||
                 'ffs_id=' || TO_CHAR(ffs_id) ||
                 '; rule_id=' || TO_CHAR(rule_id) ||
                 '; hSumLvl=' || hSumLvl ||
                 '; hExecType=' || hExecType ||
                 '; hCreateYn=' || hCreateYn ||
                 '; dailyThreshold=' || TO_CHAR(dailyThreshold) ||
                 '; premiumThreshold=' || TO_CHAR(premiumThreshold) )

/*
 * Block 4: Processing Logic
 */
MEASURE_UNDER = EMPTY_NUMBER_NUMBER
MEASURE_OVER = EMPTY_NUMBER_NUMBER
MEASURE_PREMIUM = EMPTY_NUMBER_NUMBER

tcRecordCount = HWM_CTXARY_RECORD_POSITIONS.count
max_loop = 1000

nidx = 0
WHILE (nidx < tcRecordCount) LOOP
(
  nidx = nidx + 1

  if (nidx > max_loop) then (
    ex = RAISE_ERROR (ffs_id, rule_id, 'Formula ' || ffName || ' terminated due to possible end-less loop.')
  )

  tcMeasure = 0
  tcMeasureDay = 0
  tcRecPos = NullText
  tcStartTime = NullDateTime
  tcDayKey = NullDate
  posType = NullText
  splitMeasure = 0

  if (HWM_CTXARY_RECORD_POSITIONS.exists(nidx)) then (
    tcRecPos = HWM_CTXARY_RECORD_POSITIONS[nidx]
  )
  if (measure.exists(nidx)) then (
    tcMeasure = measure[nidx]
  )
  if (HWM_CTXARY_HWM_MEASURE_DAY.exists(nidx)) then (
    tcMeasureDay = HWM_CTXARY_HWM_MEASURE_DAY[nidx]
  )
  if (StartTime.exists(nidx)) then (
    tcStartTime = StartTime[nidx]
    tcDayKey = TRUNC(tcStartTime)
  )

  if (tcRecPos = RecPositionDetail) then (
    posType = RecPositionDetail
  ) else if (tcRecPos = RecPositionEndDay) then (
    posType = RecPositionEndDay
  ) else if (tcRecPos = RecPositionEndPeriod) then (
    posType = RecPositionEndPeriod
  )

  rLog = add_rlog (ffs_id, rule_id, 'Row ' || TO_CHAR(nidx) ||
                   '; recPos=' || tcRecPos ||
                   '; posType=' || posType ||
                   '; startTime=' || TO_CHAR(tcStartTime) ||
                   '; dayKey=' || TO_CHAR(tcDayKey) ||
                   '; measure=' || TO_CHAR(tcMeasure) ||
                   '; measureDay=' || TO_CHAR(tcMeasureDay) )

  /* Default to the measure on the row. If the row is an END_DAY or END_PERIOD
   * summary row and the measure is zero, use the day measure instead.
   */
  splitMeasure = tcMeasure
  if (splitMeasure = 0 and tcMeasureDay > 0 and
      (posType = RecPositionEndDay or posType = RecPositionEndPeriod)) then (
    splitMeasure = tcMeasureDay
  )

  if (splitMeasure > 0) then (
    if (splitMeasure <= dailyThreshold) then (
      MEASURE_UNDER[nidx] = splitMeasure
      MEASURE_OVER[nidx] = 0
      MEASURE_PREMIUM[nidx] = 0
    ) else if (splitMeasure <= premiumThreshold) then (
      MEASURE_UNDER[nidx] = dailyThreshold
      MEASURE_OVER[nidx] = splitMeasure - dailyThreshold
      MEASURE_PREMIUM[nidx] = 0
    ) else (
      MEASURE_UNDER[nidx] = dailyThreshold
      MEASURE_OVER[nidx] = premiumThreshold - dailyThreshold
      MEASURE_PREMIUM[nidx] = splitMeasure - premiumThreshold
    )
  ) else (
    MEASURE_UNDER[nidx] = 0
    MEASURE_OVER[nidx] = 0
    MEASURE_PREMIUM[nidx] = 0
  )

  rLog = add_rlog (ffs_id, rule_id, 'Split Result: ' ||
                   '; splitMeasure=' || TO_CHAR(splitMeasure) ||
                   '; MEASURE_UNDER=' || TO_CHAR(MEASURE_UNDER[nidx]) ||
                   '; MEASURE_OVER=' || TO_CHAR(MEASURE_OVER[nidx]) ||
                   '; MEASURE_PREMIUM=' || TO_CHAR(MEASURE_PREMIUM[nidx]) )
)

rLog = add_rlog (ffs_id, rule_id, '<<< Exit - ' || ffName )

/*
 * Block 5: Outputs and Return
 */
RETURN MEASURE_UNDER, MEASURE_OVER, MEASURE_PREMIUM
