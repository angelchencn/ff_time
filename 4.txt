/*******************************************************************************
 * Formula Name : ORA_WFM_TCR_UNPAID_BREAK_DURATION_AP
 * Formula Type : Time Calculation Rules
 * Description  : When a schedule time entry matches a reported time entry,
 *                 subtracts the time not worked from the calculated duration.
 *
 * Detail       : The formula iterates through each time entry and checks
 *                 to see if there's a scheduled shift. When it finds
 *                 the corresponding shift, it subtracts scheduled time
 *                 not worked from the calculated time entry duration.
 *                 The formula assumes the time not worked value is measured
 *                 in minutes and converts the duration to hours.
 *
 * Change History
 * --------------
 * Who                   Ver      Date           Description
 * --------------------  -------  ------------   ---------------------------
 * Dario Vincent         1.0      2025-AUG-19    Initial development
 ******************************************************************************/

/*
 * Block 1: Default Declarations
 */
DEFAULT FOR HWM_CTXARY_RECORD_POSITIONS IS EMPTY_TEXT_NUMBER
DEFAULT FOR HWM_CTXARY_HWM_MEASURE_DAY IS EMPTY_NUMBER_NUMBER
DEFAULT FOR StartTime IS EMPTY_DATE_NUMBER
DEFAULT FOR StopTime IS EMPTY_DATE_NUMBER
DEFAULT FOR measure IS EMPTY_NUMBER_NUMBER
DEFAULT FOR UnitOfMeasure IS EMPTY_TEXT_NUMBER

DEFAULT_DATA_VALUE FOR HWM_DTL_SCHD_START_TIME IS '01-JAN-1900'(DATE)
DEFAULT_DATA_VALUE FOR HWM_DTL_SCHD_END_TIME IS '01-JAN-1900'(DATE)
DEFAULT_DATA_VALUE FOR HWM_DTL_SCHD_SCHEDULED_REF_DATE IS '01-JAN-1900'(DATE)
DEFAULT_DATA_VALUE FOR HWM_DTL_SCHD_SCHEDULED_TIME_NOT_WORKED IS 0

/*
 * Block 2: Inputs
 */
INPUTS ARE
    HWM_CTXARY_RECORD_POSITIONS,
    HWM_CTXARY_HWM_MEASURE_DAY,
    StartTime,
    StopTime,
    measure,
    UnitOfMeasure

/*
 * Block 3: Context and Parameter Setup
 */
NullDate = '01-JAN-1900'(DATE)
NullDateTime = '1900/01/01 00:00:00'(DATE)
NullText = '**FF_NULL**'
sTrue = 'Y'
sFalse = 'N'

RecPositionEndPeriod = 'END_PERIOD'
RecPositionEndDay = 'END_DAY'
RecPositionDetail = 'DETAIL'

ctx_personId = GET_CONTEXT(HWM_RESOURCE_ID, 0)
ctx_subResource = GET_CONTEXT(HWM_SUBRESOURCE_ID, 0)
ctx_SearchStartDate = GET_CONTEXT(HWM_CTX_SEARCH_START_DATE, NullDate)
ctx_SearchEndDate = GET_CONTEXT(HWM_CTX_SEARCH_END_DATE, NullDate)
ffs_id = GET_CONTEXT(HWM_FFS_ID, 0)
rule_id = GET_CONTEXT(HWM_RULE_ID, 0)

ffName = 'ORA_WFM_TCR_UNPAID_BREAK_DURATION_AP'
rLog = add_rlog(ffs_id, rule_id, '>>> Enter - ' || ffName)

pDisplayUnpaidBreak = get_rvalue_text(rule_id, 'DISPLAY_UNPAID_BREAK', 'N')

/* ----------- Rule Header parameters ----------- */
hSumLvl = Get_Hdr_Text(rule_id, 'RUN_SUMMATION_LEVEL', 'TIMECARD')
hExecType = Get_Hdr_Text(rule_id, 'RULE_EXEC_TYPE', 'CREATE')

hCreateYn = 'N'
IF (UPPER(hExecType) = 'CREATE') THEN
(
    hCreateYn = 'Y'
)

l_status = add_rlog(
    ffs_id,
    rule_id,
    'Rule Header and Context:' ||
    ' , ffs_id =' || TO_CHAR(ffs_id) ||
    ' , rule_id =' || TO_CHAR(rule_id) ||
    ' , hSumLvl=' || hSumLvl ||
    ' , hExecType=' || hExecType ||
    ' , hCreateYn=' || hCreateYn
)

/*
 * Block 4: Processing Logic
 */
tcRecordCount = HWM_CTXARY_RECORD_POSITIONS.count
hours_not_worked_measure = EMPTY_NUMBER_NUMBER
OUT_MEASURE = EMPTY_NUMBER_NUMBER

rec_idx = 0
WHILE (rec_idx < tcRecordCount) LOOP
(
    rec_idx = rec_idx + 1

    rec_start_time = NullDate
    rec_stop_time = NullDate
    rec_measure = 0
    recPos = NullText

    IF (StartTime.exists(rec_idx)) THEN
    (
        rec_start_time = StartTime[rec_idx]
    )

    IF (StopTime.exists(rec_idx)) THEN
    (
        rec_stop_time = StopTime[rec_idx]
    )

    IF (measure.exists(rec_idx)) THEN
    (
        rec_measure = measure[rec_idx]
    )

    IF (HWM_CTXARY_RECORD_POSITIONS.exists(rec_idx)) THEN
    (
        recPos = HWM_CTXARY_RECORD_POSITIONS[rec_idx]
    )

    rLog = add_rlog(
        ffs_id,
        rule_id,
        'Processing Entry ' || TO_CHAR(rec_idx) ||
        ' recPos ' || recPos ||
        ' measure ' || TO_CHAR(rec_measure) ||
        ' from ' || TO_CHAR(rec_start_time) ||
        ' to ' || TO_CHAR(rec_stop_time)
    )

    /* Change context using the time entry's start and stop times for detail position only */
    IF (recPos = RecPositionDetail) THEN
    (
        CHANGE_CONTEXTS(
            HWM_CTX_SEARCH_START_DATE = TRUNC(rec_start_time),
            HWM_CTX_SEARCH_END_DATE = ADD_DAYS(TRUNC(rec_stop_time), 1)
        )
        (
            /* Initialize output to the original measure, will be updated if a match is found */
            match_found_flag = sFalse

            sched_rec_count = HWM_DTL_SCHD_SCHEDULED_REF_DATE.count
            sched_idx = 0

            WHILE (sched_idx < sched_rec_count) LOOP
            (
                sched_idx = sched_idx + 1

                sch_start_time = HWM_DTL_SCHD_START_TIME[sched_idx]
                sch_stop_time = HWM_DTL_SCHD_END_TIME[sched_idx]

                rLog = add_rlog(
                    ffs_id,
                    rule_id,
                    'Checking Schedule Shift ' || TO_CHAR(sched_idx) ||
                    ' from ' || TO_CHAR(sch_start_time) ||
                    ' to ' || TO_CHAR(sch_stop_time) ||
                    ' pDisplayUnpaidBreak: ' || pDisplayUnpaidBreak
                )

                /* Find a scheduled shift that matches the time card entry */
                IF (
                       match_found_flag = sFalse
                   AND sch_start_time = rec_start_time
                   AND sch_stop_time = rec_stop_time
                ) THEN
                (
                    sch_time_not_worked_minutes =
                        HWM_DTL_SCHD_SCHEDULED_TIME_NOT_WORKED[sched_idx]

                    rLog = add_rlog(
                        ffs_id,
                        rule_id,
                        'Matching schedule found. Scheduled time not worked is ' ||
                        TO_CHAR(sch_time_not_worked_minutes)
                    )

                    /* Convert scheduled time not worked from minutes to hours */
                    time_not_worked_hrs = sch_time_not_worked_minutes / 60

                    OUT_MEASURE[rec_idx] = rec_measure - time_not_worked_hrs

                    IF (pDisplayUnpaidBreak = 'Y' AND time_not_worked_hrs > 0) THEN
                    (
                        hours_not_worked_measure[rec_idx] = time_not_worked_hrs

                        rLog = add_rlog(
                            ffs_id,
                            rule_id,
                            'Unpaid break measure is ' ||
                            TO_CHAR(hours_not_worked_measure[rec_idx])
                        )
                    )

                    match_found_flag = sTrue

                    rLog = add_rlog(
                        ffs_id,
                        rule_id,
                        'New measure is ' || TO_CHAR(OUT_MEASURE[rec_idx])
                    )
                )
            )

            rLog = add_rlog(
                ffs_id,
                rule_id,
                'match_found_flag ' || match_found_flag
            )
        )
    )
)

rLog = add_rlog(ffs_id, rule_id, '<<< Exit - ' || ffName)

/*
 * Block 5: Outputs
 */
RETURN OUT_MEASURE,
       hours_not_worked_measure
